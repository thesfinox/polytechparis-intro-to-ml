\begin{frame}{Unsupervised ML}{The one with no labels\dots}
    Let us start by considering \textbf{unsupervised learning}:
    \begin{equation*}
        \mathcal{D} = \left\{ \vec{x}_i \right\}_{i = 1}^N
    \end{equation*}

    \vfill

    \begin{columns}[c]
        \begin{column}{0.5\textwidth}
            Generally used for:
            \begin{itemize}
                \item \alert<2>{Data \textbf{exploration}: useful to see if data have particular properties}
                \item \alert<3>{Finding \textbf{anomalous} behaviour: can be used to find \emph{outliers}}
                \item \alert<4>{Identification of common \textbf{characteristics}: used to identify \enquote{similar} samples}
            \end{itemize}
        \end{column}
        \hfil

\begin{frame}{KMeans Clustering}{Grouping based on proximity\dots}
    The first algorithm is \textbf{KMeans} clustering:

    \vfill

    \begin{columns}[c]
        \begin{column}{0.6\textwidth}
            \alert<2>{\textbf{Start:} choose a number of \enquote{groups} $K$ and choose their centre randomly}

            \textbf{Loop N times:}
            \begin{enumerate}
                \item \alert<3>{Assign samples to the nearest group centre}
                \item \alert<4>{Use samples in the group to compute again the centre}
                \item \textbf{After N iterations:} stop and perform a final assignment of samples
            \end{enumerate}
        \end{column}
        \hfil
        \begin{column}{0.4\textwidth}
            \centering
            \input{03_slides/img/kmeans.tex}
        \end{column}
    \end{columns}

    \vfill

    The procedure is based on \textbf{Euclidean distance}: $d(\vec{x}, \vec{c}) = \norm{\vec{x} - \vec{c}}$.
    It can be \textbf{trained} on a set and \enquote{applied} on another!
\end{frame}

\begin{frame}{DBSCAN}{Grouping based on density\dots}
    The second algorithm we see is \textbf{DBSCAN}:

    \vfill

    \begin{columns}[c]
        \begin{column}{0.6\textwidth}
            \alert<2>{\textbf{Start:} Set the number of clusters to $0$}

            \textbf{Loop over all points P:}
            \begin{enumerate}
                \item if P was already considered then skip
                \item \alert<3>{select P and assign it to a \enquote{group} (\emph{cluster})}
                \item \alert<4>{\textbf{Loop over all points Q near P:}}
                      \begin{enumerate}
                          \item \alert<5>{select Q and add it to cluster}
                          \item \alert<6-7>{Extend the cluster with neighbours of Q}
                          \item \alert<8>{Close the cluster when no more points are unexplored in the vicinity}
                      \end{enumerate}
            \end{enumerate}
        \end{column}
        \hfil
        \begin{column}{0.4\textwidth}
            \centering
            \input{03_slides/img/dbscan.tex}
        \end{column}
    \end{columns}

    \vfill

    \textbf{Key difference:} You do no need to set the number of clusters to start! Only based on \enquote{density}: does \textbf{not transfer} to different sets
\end{frame}

\begin{frame}{Clustering}{Is it like classification?}
    Clustering creates \textbf{pseudo-labels}:
    \begin{center}
        \begin{tabular}{@{}cccc@{}}
            $x_1$ & $x_2$ & $x_3$ & $K$   \\ \hline
            0.3   & 0.1   & 0.4   & 1     \\
            -0.3  & 0.14  & 0.24  & 2     \\
            -0.4  & 0.12  & 0.26  & 2     \\
            \dots & \dots & \dots & \dots
        \end{tabular}
    \end{center}

    \vfill \pause

    However, we cannot score it like classification.
    Most used metrics:
    \begin{itemize}
        \item if \textbf{ground truth} is available $\rightarrow$ \emph{Rand score}:
              \begin{equation*}
                  \operatorname{Rand} = \frac{\text{no.\ of matching pairs}}{\text{total pairs}}
              \end{equation*}
        \item if \textbf{no ground truth} is available $\rightarrow$ \emph{Silhouette score}:
              \begin{equation*}
                  \operatorname{S} = \frac{1}{N} \sum\limits_{i = 1}^N \frac{b_i - a_i}{\max\{ a_i, b_i \}},
                  \quad
                  a_i = \mathbb{E}_{i \neq j \in C_i}\left[ \operatorname{dist}(i, j) \right],
                  \quad
                  b_i = \mathbb{E}_{i \neq j \in \text{nearest~}C_j}\left[ \operatorname{dist}(i, j) \right]
              \end{equation*}
    \end{itemize}
\end{frame}

\begin{frame}{Visualisation in High Dimension}{What to do with large data?}
    The principal idea is then to find:
    \begin{equation*}
        f\colon \quad \mathbb{R}^d \longrightarrow \mathbb{R}^{p}
    \end{equation*}
    where
    \begin{equation*}
        p \ll d
    \end{equation*}
    and ideally $p = 2, 3$.
    This way,
    \begin{equation*}
        \vec{x}^{\prime} = f(\vec{x})
    \end{equation*}
    would become \enquote{manageable}.

    \vfill \pause

    What \textbf{properties} must we require?
    We will mainly see:
    \begin{itemize}
        \item (PCA) If data contains \enquote{signal}, rotate it in the right direction
        \item (tSNE) Preserve the concept of \enquote{distance} in high and low dimensions
    \end{itemize}
\end{frame}

\begin{frame}{Principal Components Analysis}{Finding the direction of signal\dots}
    \begin{block}{Prerequisite 1: the Spectral Theorem}
        Let $C$ be a square symmetric real matrix $C = C^{\top} \in \mathbb{R}^{d \times d}$, then
        \begin{equation*}
            \exists \Lambda = \operatorname{diag}\left( \lambda_1, \lambda_2, \dots, \lambda_d \right)
        \end{equation*}
        where $\lambda_i$ are \textbf{eigenvalues} of $C$, and
        \begin{equation*}
            W = \left( \vec{w}_i^{\top} \right)_{i = 1}^d,
            \quad
            \vec{w}_i = \left( w_{i1}~w_{i2}~\dots~w_{id} \right)
        \end{equation*}
        is the matrix of associated \textbf{eigenvectors} (as columns), such that
        \begin{equation*}
            C = W \Lambda W^{\top}.
        \end{equation*}
    \end{block}

    \vfill

    In other words:
    \begin{equation*}
        C = \sum\limits_{i = 1}^d \lambda_i \vec{w}_i^T \vec{w}_i
    \end{equation*}
\end{frame}

\begin{frame}{Principal Components Analysis}{\includegraphics[height=12pt]{01_slides/img/tool.pdf}~Finding the direction of signal\dots}
    Consider some data such that $\mathbb{E}[X] = 0$:
    \begin{equation*}
        X = \left( \mathbb{I}_{n \times p} + \beta \vec{v}^{\top} \vec{u} \right)^{\frac{1}{2}} Z \simeq
        \left( \mathbb{I}_{n \times p} + \frac{1}{2} \beta \vec{v}^{\top} \vec{u} \right) Z \in \mathbb{R}^{n \times p}
    \end{equation*}
    where $\norm{\vec{u}} = \norm{\vec{v}} = 1$ and $\beta > 0$, and $Z = \mathbb{I}_{p \times p} + \frac{1}{\sqrt{n}} W$, and $W \sim \mathcal{N}(O, \mathbb{I}_{p \times p})$.

    \vfill \pause

    \includegraphics[height=12pt]{01_slides/img/pen-tool.pdf}~Let us build the \textbf{covariance matrix} (symmetric):
    \vfill \pause
    \begin{onlyenv}<2-3>
        \begin{equation*}
            \begin{split}
                C
                =
                \frac{1}{n} X^{\top} X
                & =
                Z^{\top}
                (
                \mathbb{I}_{p \times p}
                +
                \beta
                \overbrace{\frac{\vec{u}^{\top} \vec{v}\,  \mathbb{I}_{n \times p} + \mathbb{I}_{p \times n}\, \vec{v}^{\top} \vec{u}}{2}}^{A = A^T \in \mathbb{R}^{p \times p}}
                ) Z
                +
                \mathcal{O}(\beta^2)
                \\
                & \simeq
                \mathbb{I}_{p \times p}
                +
                \beta A
                +
                \frac{1}{\sqrt{n}} \left( W^{\top} + W \right)
                \\
                & =
                \underbrace{%
                    \beta A
                    +
                    \widetilde{W}
                }_{\text{all symmetric!}}
                \only<3>{%
                \alert{%
                =
                \beta
                \vec{a}^{\top} \vec{a}
                +
                \sum\limits_{i = 1}^d
                \tilde{\lambda}_i
                \tilde{\vec{w}}_i^{\top}
                \tilde{\vec{w}}_i
                }
                }
            \end{split}
        \end{equation*}
    \end{onlyenv}
    \begin{onlyenv}<4>
        \begin{equation*}
            C
            =
            \beta \vec{a}^{\top} \vec{a}
            +
            \sum\limits_{i = 1}^d \tilde{\lambda}_i \tilde{\vec{w}}_i^{\top} \tilde{\vec{w}}_i
            =
            \sum\limits_{i = 1}^d \lambda_i \vec{w}_i^{\top} \vec{w}_i,
        \end{equation*}
        where $\lambda_1 = \beta + \tilde{\lambda}_1$, and $\lambda_1 > \lambda_2 > \dots > \lambda_d$.
    \end{onlyenv}
\end{frame}

\begin{frame}{Principal Components Analysis}{Finding the direction of signal\dots}
    We need a way to find $\beta$ (though we do not know \emph{where} it is  and \emph{if} we can see it):
    \begin{equation*}
        C
        =
        \sum\limits_{i = 1}^d \lambda_i \vec{w}_i^{\top} \vec{w}_i,
        \gets
        \beta~\text{is somewhere there}
    \end{equation*}
    which means that we need to look for a \textbf{transformation}
    \begin{equation*}
        X^{\prime} = X\, T,
        \quad
        T = T^{\top} \in \mathbb{R}^{p \times p}
    \end{equation*}
    such that
    \begin{equation*}
        \vec{t}_1
        =
        \arg\max\limits_{\vec{t} \mid \norm{t} = 1}
        \operatorname{Var}(\vec{x}^{\prime}_1)
        =
        \arg\max\limits_{\vec{t}}
        \frac{\vec{t}^{\,\top} X^{\top} X \vec{t}}{\vec{t}^{\top} \vec{t}}
        =
        \arg\max\limits_{\vec{t}}
        \frac{\vec{t}^{\,\top} C \vec{t}}{\vec{t}^{\top} \vec{t}},
    \end{equation*}
    and so on\dots

    \vfill \pause

    \begin{center}
        \textbf{Property:} $\vec{t}_i$ ($i = 1, \dots, d$) must be \textbf{eigenvectors} of $C$.
    \end{center}

    \vfill

    {\tiny
        Clearly, this is only going to be possible if $\beta$ is large enough to be detected!
    }
\end{frame}

\begin{frame}{Principal Components Analysis}{Finding the direction of signal\dots}
    We thus find that
    \begin{equation*}
        \operatorname{Var}(\vec{t}_i)
        =
        \vec{t}_i^{\,\top} C \vec{t}_i
        =
        \lambda_i
    \end{equation*}
    and in particular \emph{if good conditions are met}:
    \begin{equation*}
        \operatorname{Var}(\vec{t}_1)
        =
        \vec{t}_1^{\,\top} C \vec{t}_1
        =
        \alert{\beta + \tilde{\lambda}_i}
    \end{equation*}

    \vfill \pause

    \begin{center}
        \huge
        We found the direction ($\vec{t}_1$) of the signal!
    \end{center}

    \vfill \pause

    This is called called \textbf{Principal Components Analysis} (PCA):
    \begin{equation*}
        \underbrace{~~X^{\prime}~~}_{\text{scores/principal components}} = X \underbrace{~~T~~}_{\text{loadings}}.
    \end{equation*}
    They enable \textbf{ranking} the content of the data, according to their \textbf{variance} (\emph{importance})
\end{frame}

\begin{frame}{Principal Components Analysis}{Reducing the dimensionality of the data\dots}
    PCA has an interesting byproduct:
    \begin{center}
        \input{03_slides/img/pca.tex}
    \end{center}
    \begin{enumerate}
        \item<1-> Suppose you have data $X \in \mathbb{R}^{n \times p}$ and find $T \in \mathbb{R}^{p \times p}$ for PCA
        \item<2-> Then, naturally, $X^{\prime} \in \mathbb{R}^{n \times p}$
        \item<3-> However, the important part is concentrated in few components of $T_{a} \preceq T$, $a < p$.
        \item<4-> Hence, the product $\alert{X^{\prime}_a} = X \alert{T_a \in \mathbb{R}^{n \times a}}$ has a \textbf{reduced dimensionality}, but not \textbf{reduced information}!
    \end{enumerate}
\end{frame}

\begin{frame}{Principal Components Analysis}{Reducing the dimensionality of the data\dots}
    PCA allows you to choose a subset of components from your data, losing as little information as possible:
    \vfill
    \begin{columns}[c]
        \begin{column}{0.6\textwidth}
            The eigenvalues of the covariance matrix define the \textbf{explained variance} of each component:
            \begin{enumerate}
                \item \alert<1>{You can select all components that are \textbf{higher} than a certain threshold (e.g.\ $\lambda_i > 5\%$)}
                \item \alert<2>{You can select all components that \textbf{cumulatively} add up to a certain variance threshold (e.g.\ $\sum_{i < a} \lambda_i \le 95\%$)}
            \end{enumerate}
        \end{column}

\end{frame}

\begin{frame}{Visualisation Tools}{Keeping the distance\dots}
    Another \textbf{dimensionality reduction} technique is
    \vfill
    \begin{center}
        \large
        \emph{t-distributed Stochastic Neighbours Embedding} (tSNE)
    \end{center}
    \vfill
    The idea is the following:
    \begin{columns}[c]
        \begin{column}{0.5\textwidth}
            \begin{itemize}
                \item You need to find \emph{mathematical objects} that do not change in high- and low-dimension
                \item You need something that relates data (something \emph{relative})
            \end{itemize}

            \includegraphics[height=12pt]{01_slides/img/brain.pdf}~What kind of objects can we use? What property do they have?
        \end{column}
        \hfil
        \only<2>{
            \begin{column}{0.5\textwidth}
                \begin{center}
                    \large
                    We use \textbf{distances} between points!

                    They are scalars and do not transform differently in different dimensions!
                \end{center}
            \end{column}
        }
        \only<3>{
            \begin{column}<3>{0.4\textwidth}
                \begin{center}
                    \large
                    \textbf{Key idea:} maintain the same \emph{distance} value in high dimension and low dimension!
                \end{center}
            \end{column}
        }
    \end{columns}
\end{frame}

\begin{frame}{Visualisation Tools}{Keeping the distance\dots}
    \vfill
    \begin{columns}[T]
        \begin{column}<1->{0.48\textwidth}
            Consider two \emph{existing} $\vec{x}_i \in \mathbb{R}^D$ and $\vec{x}_j \in \mathbb{R}^D$:
            \begin{center}
                \large \textbf{HIGH DIMENSION}
            \end{center}
            Compute a \enquote{distance probability}:
            \begin{equation*}
                P(\vec{x}_i, \vec{x}_j)
                =
                \frac{
                    e^{-\frac{\norm{\vec{x}_i - \vec{x}_j}^2}{2 \sigma_i^2}}
                }{
                    \sum\limits_{i \neq j}
                    e^{-\frac{\norm{\vec{x}_i - \vec{x}_j}^2}{2 \sigma_i^2}}
                }
                +
                (i \leftrightarrow j)
            \end{equation*}

            This is a \textbf{Gaussian} distribution over distances.
        \end{column}
        \hfil
        \begin{column}<2->{0.48\textwidth}
            Consider two \emph{images} $\vec{z}_i \in \mathbb{R}^d$ and $\vec{z}_j \in \mathbb{R}^d$:
            \begin{center}
                \large \textbf{LOW DIMENSION}
                Compute a \emph{desired} \enquote{distance probability}:
                \begin{equation*}
                    Q(\vec{x}_i, \vec{x}_j)
                    =
                    \frac{
                        \left( 1 + \norm{\vec{z}_i - \vec{z}_j}^2 \right)^{-1}
                    }{
                        \sum\limits_{i \neq j}
                        \left( 1 + \norm{\vec{z}_i - \vec{z}_j}^2 \right)^{-1}
                    }
                    +
                    (i \leftrightarrow j)
                \end{equation*}

                This is a \textbf{Cauchy} distribution over distances.
            \end{center}
        \end{column}
    \end{columns}
    \vfill
\end{frame}

\begin{frame}{Visualisation Tools}{Keeping the distance\dots}
    What is the advantage of using this?

    \vfill \pause

    Using \textbf{gradient descent}, we can minimise the distance between the two distributions:
    \vfill
    \begin{columns}[c]
        \begin{column}{0.4\textwidth}
            \begin{equation*}
                \begin{split}
                    \operatorname{dist}(P, Q)
                    & =
                    \mathrm{D}_{\text{KL}}\left( P \Vert Q \right)
                    \\
                    & =
                    \frac{1}{N}
                    \sum\limits_{i, j}
                    P(\vec{x}_i, \vec{x}_j)
                    \ln
                    \frac{P(\vec{x}_i, \vec{x}_j)}{Q(\vec{x}_i, \vec{x}_j)}
                \end{split}
            \end{equation*}
        \end{column}
